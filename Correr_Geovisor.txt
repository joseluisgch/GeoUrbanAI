# ========================================================
# PASOS PARA EJECUTAR GEOVISOR IA EN LOCAL
# ========================================================

# --------------------------------------------------------
# TERMINAL 1: Levantar el Backend (FastAPI con Gemini)
# --------------------------------------------------------
cd E:\geovisor-ia\backend
python -m uvicorn main:app --reload --port 8000

# Verificar si la API responde en el navegador:
# http://localhost:8000/health
# http://localhost:8000/docs


# --------------------------------------------------------
# TERMINAL 2: Levantar el Frontend (Servidor Web)
# --------------------------------------------------------
cd E:\geovisor-ia\frontend
python -m http.server 3000

# Abrir en el navegador:
# http://localhost:3000

# Nota: El frontend detectará automáticamente que estás en local
# y se conectará a http://localhost:8000 (indicador verde "API conectada").
